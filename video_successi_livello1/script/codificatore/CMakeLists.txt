cmake_minimum_required(VERSION 3.10)
project(codifica_h264 CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ignition-common4 REQUIRED COMPONENTS graphics)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FF REQUIRED libavcodec libavutil libswscale)
add_executable(codifica_h264 codifica_h264.cc)
target_include_directories(codifica_h264 PRIVATE ${FF_INCLUDE_DIRS})
target_link_libraries(codifica_h264 ignition-common4::core ignition-common4::graphics ${FF_LIBRARIES})
